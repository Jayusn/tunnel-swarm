cmake_minimum_required(VERSION 3.0.2)
project(mot_mapping)
find_package(catkin REQUIRED COMPONENTS roscpp pcl_conversions message_filters sensor_msgs nav_msgs geometry_msgs visualization_msgs obj_state_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(Threads REQUIRED)
catkin_package(CATKIN_DEPENDS roscpp obj_state_msgs)
add_executable(mapping_node src/mapping_node.cpp src/mapping_manger.cpp src/Hungarian.cpp)
target_compile_features(mapping_node PRIVATE cxx_std_14)
set_target_properties(mapping_node PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(mapping_node PRIVATE include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
target_link_libraries(mapping_node ${catkin_LIBRARIES} ${PCL_LIBRARIES} Threads::Threads)
add_dependencies(mapping_node ${catkin_EXPORTED_TARGETS})

if(CATKIN_ENABLE_TESTING)
  add_executable(density_partition_test test/density_partition_test.cpp)
  target_compile_features(density_partition_test PRIVATE cxx_std_14)
  target_include_directories(density_partition_test PRIVATE include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
  target_link_libraries(density_partition_test ${PCL_LIBRARIES})
  add_test(NAME mot_mapping_density_partition COMMAND density_partition_test)
endif()
